\documentclass[tikz,border=3pt]{standalone}

% Beam states from the "Avoiding dead ends" slide: prune, clone, step.
% Each row is one particle drawn as its vertices. Slide coordinates and sizes;
% the slide's colours are rendered as plain black and grey.
% The macros live in the body, not the preamble, so that \includestandalone
% in the paper (default mode) picks them up together with the picture.

\begin{document}

\newcommand{\gap}{0.3}
\newcommand{\midy}{-0.93} % vertical centre of the four rows

% \dotsrow{x0}{y}{n}{colour}: one particle as n vertices
\newcommand{\dotsrow}[4]{%
	\foreach \i in {1,...,#3}{%
		\node[dot, fill=#4] at ({#1+\i*\gap-\gap},#2) {};%
	}%
}

% \strike{x0}{y}{n}: struck-through particle, as for a stalled one
\newcommand{\strike}[3]{\draw[line width=1.2pt] ({#1-0.25},#2) -- ({#1+\gap*#3-0.05},#2);}

% \clonebox{x0}{y}{n}: refilled slot, as for a copy of a survivor
\newcommand{\clonebox}[3]{\draw[dashed, line width=0.8pt, rounded corners=0.08cm] ({#1-0.2},{#2-0.2}) rectangle ({#1+\gap*#3-0.1},{#2+0.2});}

% \beamarrow{x}: the slide's arrow between two beam states
\newcommand{\beamarrow}[1]{\draw[->, line width=0.6pt, black!50] ({#1-0.5},\midy) -- ({#1+0.5},\midy);}


\begin{tikzpicture}[
	dot/.style={circle, inner sep=0pt, minimum size=0.18cm},
	note/.style={anchor=west, font=\small},
]

	% ---- prune: stalled particle and unreachable density band --------------
	\dotsrow{0}{0}{6}{black}
	\dotsrow{0}{-0.62}{9}{black!25}
	\strike{0}{-0.62}{9}
	\node[note] at (2.85,-0.62) {stalled};
	\dotsrow{0}{-1.24}{8}{black!25}
	\strike{0}{-1.24}{8}
	\node[note] at (2.55,-1.24) {density};
	\dotsrow{0}{-1.86}{8}{black}

	\beamarrow{4.55}

	% ---- clone: empty slots refilled from survivors ------------------------
	\dotsrow{5.4}{0}{6}{black}
	\clonebox{5.4}{-0.62}{8}
	\dotsrow{5.4}{-0.62}{8}{black}
	\node[note] at (7.95,-0.62) {copy};
	\clonebox{5.4}{-1.24}{6}
	\dotsrow{5.4}{-1.24}{6}{black}
	\node[note] at (7.35,-1.24) {copy};
	\dotsrow{5.4}{-1.86}{8}{black}

	\beamarrow{9.45}

	% ---- step: the least-grown particle takes one legal extension ----------
	\dotsrow{10.4}{0}{7}{black}
	\draw[line width=0.6pt] (12.2,0) circle (0.16);
	\node[note] at (12.65,0) {+z};
	\dotsrow{10.4}{-0.62}{8}{black}
	\dotsrow{10.4}{-1.24}{6}{black}
	\dotsrow{10.4}{-1.86}{8}{black}

\end{tikzpicture}

\end{document}
